## 📌 1. Arquitectura del Monitoreo Local

Este sistema permite ejecutar diagnósticos automáticos y periódicos desde tu laptop de desarrollo (Windows) hacia los dos servidores VPS en producción:

1. **VPS 1 (Contabo)**: `169.58.168.107`
   - **Stack**: Docker, Coolify, Supabase, Caddy, APEFAC, InAndes ERP (Nuevo), FastAPI Worker PDF.
2. **VPS 2 (Hostinger)**: `91.108.125.253`
   - **Stack**: PETRAL SMART DASHBOARD (`geeksoft-engine`), Nginx, FastAPI, Servicios Legacy.

El script se conecta por SSH seguro vía `Paramiko`, ejecuta las sondas de telemetría sin instalar agentes pesados en los servidores, audita la salud de hardware, **verifica la conectividad interna Caddy -> FastAPI (Worker PDF)**, valida la persistencia de alias de red en Coolify DB y guarda un log consolidado diario con alertas visuales.

Ubicación del script en tu laptop: `C:\Users\rguti\Inandes.ERP.React\scripts\monitor_vps_daily.py` (o `C:\Users\rguti\APEFAC\scripts\monitor_vps_daily.py`)

```python
import os
import sys
import datetime
import paramiko

sys.stdout.reconfigure(encoding='utf-8', errors='replace')

# ==============================================================================
# CONFIGURACIÓN DE SERVIDORES Y SERVICIOS
# ==============================================================================
SERVERS = [
    {
        "name": "VPS Contabo (InAndes ERP / Coolify / APEFAC / Supabase)",
        "host": "169.58.168.107",
        "port": 22,
        "user": "root",
        "pass": "VivaLaVida2026$",
        "is_docker": True,
        "fastapi_alias": "3g5kcala3ypqzlsrhyelxyev",
        "fastapi_port": 8010
    },
    {
        "name": "VPS Hostinger (PETRAL SMART DASHBOARD / Geeksoft Engine)",
        "host": "91.108.125.253",
        "port": 22,
        "user": "root",
        "pass": "Thiagutz061121@",
        "is_docker": False
    }
]

LOG_DIR = r"C:\Users\rguti\VPS_Health_Logs"
os.makedirs(LOG_DIR, exist_ok=True)
TODAY_STR = datetime.datetime.now().strftime("%Y-%m-%d")
LOG_FILE = os.path.join(LOG_DIR, f"vps_health_{TODAY_STR}.log")

# ==============================================================================
# FUNCIONES AUXILIARES DE EJECUCIÓN SSH
# ==============================================================================
def ssh_exec(client, command, timeout=25):
    try:
        stdin, stdout, stderr = client.exec_command(command, timeout=timeout)
        out = stdout.read().decode('utf-8', errors='replace').strip()
        err = stderr.read().decode('utf-8', errors='replace').strip()
        return out, err
    except Exception as e:
        return "", str(e)

def audit_server(srv):
    print("=" * 85)
    print(f"🛰️  INICIANDO DIAGNÓSTICO PROFUNDO: {srv['name']} ({srv['host']})")
    print(f"📅  Fecha y Hora Local: {datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print("=" * 85)

    client = paramiko.SSHClient()
    client.set_missing_host_key_policy(paramiko.AutoAddPolicy())

    try:
        client.connect(
            hostname=srv['host'],
            port=srv['port'],
            username=srv['user'],
            password=srv['pass'],
            timeout=15
        )
        print(">> Conexión SSH establecida con éxito ✓\n")

        # 1. Uptime y Carga de CPU
        uptime_out, _ = ssh_exec(client, "uptime")
        cores_out, _ = ssh_exec(client, "nproc")
        print(f"📌 [1] Uptime & Load: {uptime_out}")
        print(f"    Cores CPU disponibles: {cores_out}\n")

        # 2. Memoria RAM y SWAP
        mem_out, _ = ssh_exec(client, "free -m -h")
        print("📌 [2] Uso de Memoria RAM & SWAP:")
        print(mem_out + "\n")

        # 3. Espacio en Disco
        disk_out, _ = ssh_exec(client, "df -h /")
        print("📌 [3] Espacio en Disco (Partición Root):")
        print(disk_out + "\n")

        # 4. Top Procesos por RAM y CPU
        top_mem, _ = ssh_exec(client, "ps aux --sort=-%mem | head -n 6")
        print("📌 [4] Top 5 Procesos por Consumo de RAM (%):")
        print(top_mem + "\n")

        top_cpu, _ = ssh_exec(client, "ps aux --sort=-%cpu | head -n 6")
        print("📌 [5] Top 5 Procesos por Consumo de CPU (%):")
        print(top_cpu + "\n")

        # 6. Servicios del Sistema en Estado FAILED
        failed_units, _ = ssh_exec(client, "systemctl --failed --no-pager")
        print("📌 [6] Servicios del Sistema en Estado FAILED:")
        if "0 loaded units listed" in failed_units or not failed_units:
            print("    🟢 Ningún servicio en estado fallido.\n")
        else:
            print("    🔴 " + failed_units + "\n")

        # 7. Diagnósticos Específicos por Servidor
        if srv.get("is_docker"):
            # A) Contenedores Docker Activos
            docker_stats, _ = ssh_exec(client, "docker stats --no-stream --format 'table {{.Name}}\t{{.CPUPerc}}\t{{.MemUsage}}\t{{.MemPerc}}' | head -n 12")
            print("📌 [7] Estado de Contenedores Docker Principales:")
            print(docker_stats + "\n")

            # B) SONDA DE SALUD: WORKER DE PDF & BACKEND FASTAPI (CONTABO)
            print("=" * 85)
            print("🩺 [8] SONDA CRÍTICA: WORKER DE GENERACIÓN DE PDF & BACKEND FASTAPI")
            print("=" * 85)

            # Sonda 1: Contenedor Backend corriendo
            backend_cid, _ = ssh_exec(client, "docker ps -q --filter 'name=3g5kcala3ypqzlsrhyelxyev'")
            if backend_cid:
                print(f"    🟢 Contenedor Backend FastAPI: ACTIVO (ID: {backend_cid[:12]})")
                # Test de respuesta local de Uvicorn
                uvicorn_code, _ = ssh_exec(client, f"docker exec {backend_cid} curl -s -o /dev/null -w '%{{http_code}}' http://127.0.0.1:{srv['fastapi_port']}/docs")
                print(f"    🟢 Uvicorn en Puerto Interno {srv['fastapi_port']}: HTTP {uvicorn_code} (Operativo)")
            else:
                print("    🔴 ALERTA CRÍTICA: Contenedor Backend FastAPI NO encontrado o detenido!")

            # Sonda 2: Conectividad y Resolución de Alias DNS Interno Caddy -> FastAPI
            frontend_cid, _ = ssh_exec(client, "docker ps -q --filter 'name=yjttbctaekty8zb5ode0hiu4'")
            if frontend_cid:
                dns_alias_code, _ = ssh_exec(client, f"docker exec {frontend_cid} curl -s -o /dev/null -w '%{{http_code}}' http://{srv['fastapi_alias']}:{srv['fastapi_port']}/openapi.json")
                if dns_alias_code == "200":
                    print(f"    🟢 Canal Interno Caddy -> http://{srv['fastapi_alias']}:{srv['fastapi_port']}: HTTP 200 (Alias DNS OK)")
                else:
                    print(f"    🔴 ALERTA: Fallo de enlace interno DNS en Caddy hacia {srv['fastapi_alias']}: HTTP {dns_alias_code} (Riesgo 502 en PDF)")
            else:
                print("    🔴 ALERTA CRÍTICA: Contenedor Frontend InAndes NO encontrado!")

            # Sonda 3: Verificación de Persistencia en Base de Datos Coolify
            coolify_alias_check, _ = ssh_exec(client, "docker exec coolify-db psql -U coolify -d coolify -t -A -c \"SELECT custom_network_aliases FROM applications WHERE id = 2;\"")
            if srv['fastapi_alias'] in coolify_alias_check:
                print(f"    🟢 Persistencia Coolify DB (Application ID 2): custom_network_aliases = '{coolify_alias_check}' (Blindado contra redeploys)")
            else:
                print(f"    🟡 ADVERTENCIA: Coolify DB no tiene registrado el alias '{srv['fastapi_alias']}'. Valor actual: '{coolify_alias_check}'")

            # Sonda 4: Verificación de Endpoints Públicos SSL
            print("\n📌 [9] Verificación de URLs Públicas de Producción (HTTPS):")
            inandes_web_code, _ = ssh_exec(client, "curl -k -s -o /dev/null -w '%{http_code}' https://inandes.geeksoft.tech/")
            pdf_endpoint_code, _ = ssh_exec(client, "curl -k -s -o /dev/null -w '%{http_code}' -X POST https://inandes.geeksoft.tech/api/inversionistas/generate-pdf")
            apefac_web_code, _ = ssh_exec(client, "curl -k -s -o /dev/null -w '%{http_code}' https://apefac.geeksoft.tech/")

            print(f"    - InAndes Frontend (https://inandes.geeksoft.tech/): HTTP {inandes_web_code} {'🟢' if inandes_web_code == '200' else '🔴'}")
            print(f"    - InAndes Worker PDF (https://inandes.geeksoft.tech/api/...): HTTP {pdf_endpoint_code} {'🟢 (Ruta responde)' if pdf_endpoint_code in ['422', '200', '400'] else '🔴 (Error 502/503)'}")
            print(f"    - APEFAC Frontend (https://apefac.geeksoft.tech/): HTTP {apefac_web_code} {'🟢' if apefac_web_code == '200' else '🔴'}\n")

        else:
            # Diagnóstico Hostinger (Systemd + Nginx)
            petral_svc, _ = ssh_exec(client, "systemctl is-active geeksoft-engine")
            nginx_svc, _ = ssh_exec(client, "systemctl is-active nginx")
            print(f"📌 [7] Estado Servicios Core Hostinger:")
            print(f"    - geeksoft-engine (FastAPI): {petral_svc.upper()} {'🟢' if petral_svc == 'active' else '🔴'}")
            print(f"    - nginx: {nginx_svc.upper()} {'🟢' if nginx_svc == 'active' else '🔴'}\n")

    except Exception as e:
        print(f"❌ ERROR al conectar o auditar {srv['name']}: {e}\n")
    finally:
        client.close()

# ==============================================================================
# EJECUCIÓN PRINCIPAL CON VOLCADO A LOG
# ==============================================================================
def main():
    class Logger(object):
        def __init__(self, filename):
            self.terminal = sys.stdout
            self.log = open(filename, "a", encoding="utf-8")
        def write(self, message):
            self.terminal.write(message)
            self.log.write(message)
        def flush(self):
            self.terminal.flush()
            self.log.flush()

    sys.stdout = Logger(LOG_FILE)
    print(f"\n=====================================================================================")
    print(f"📋 REPORTE DE TELEMETRÍA Y SALUD FORENSE DE SERVIDORES VPS (CONTABO & HOSTINGER)")
    print(f"🕒 Inicio: {datetime.datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print(f"=====================================================================================\n")

    for server in SERVERS:
        audit_server(server)

    print(f"✅ Diagnóstico completo finalizado. Log consolidado en: {LOG_FILE}\n")

if __name__ == "__main__":
    main()
```

## 🚨 4. Matriz de Umbrales Críticos para Diagnóstico Rápido

Al revisar los resultados en el log o terminal, guíate con esta tabla de control:

| Recurso / Componente | Nivel Normal (🟢) | Advertencia (🟡) | Peligro / Acción Inmediata (🔴) |
| :--- | :--- | :--- | :--- |
| **Worker PDF (FastAPI Backend)** | `HTTP 200` en puerto `8010` | Uvicorn reiniciando | Contenedor caído / detenido |
| **Canal Caddy -> FastAPI (Alias DNS)** | `HTTP 200` en `http://3g5kcala3ypqzlsrhyelxyev:8010` | Latencia > 2s | `HTTP 502 / lookup failed` (Falta alias de red) |
| **Persistencia Coolify DB** | `custom_network_aliases = '3g5kcala3ypqzlsrhyelxyev'` | Campo vacío | Ejecutar SQL de re-inyección de alias |
| **Rutas Públicas HTTPS** | `HTTP 200` (Web) / `HTTP 422/200` (API PDF) | `HTTP 301/308` | `HTTP 502 / 503` (Proxy o Backend caído) |
| **RAM Usada (Contabo 12 GB)** | `< 6 GB` (50%) | `6 GB - 9 GB` (50%-75%) | `> 10 GB` (Reiniciar servicios con fugas) |
| **RAM Usada (Hostinger 4 GB)** | `< 2.5 GB` (62%) | `2.5 GB - 3.2 GB` (80%) | `> 3.5 GB` (Peligro OOM Killer) |
| **CPU Load Average** | `< Número de Cores` | `= Número de Cores` | `> 2x Cores` (CPU al 100% saturada) |
| **Espacio en Disco (Root `/`)**| `< 70%` | `70% - 85%` | `> 90%` (Limpiar logs/docker prune) |
| **Servicios Systemd** | `0 failed` | Unidades network en boot | Servicios de backend caídos (`inactive`/`failed`) |
| **Contenedores Docker** | CPU `< 20%` cada uno | CPU `20% - 60%` | CPU `> 80%` o reinicios continuos |
